<numr-calculator> Web Component

A portable calculator widget. Embed anywhere with a single import. 

<script type="module" src="numr-calculator.js?v=bfe0d8e37f"></script>
<numr-calculator></numr-calculator>
Default
Minimal (no header/footer): <numr-calculator minimal>
Custom Theme (Catppuccin Mocha via CSS variables)
Readonly: <numr-calculator readonly>
Autosave: <numr-calculator autosave>

Usage Examples

// Basic embed
<numr-calculator></numr-calculator>

// With initial content
<numr-calculator content="100 + 50\n* 2"></numr-calculator>

// Minimal mode (no header/footer)
<numr-calculator minimal></numr-calculator>

// Readonly (for embedding examples)
<numr-calculator readonly></numr-calculator>

// Skip exchange rate fetching
<numr-calculator norates></numr-calculator>

// Auto-save to localStorage
<numr-calculator autosave></numr-calculator>

// Multiple instances with separate storage
<numr-calculator autosave storage-key="calc-1"></numr-calculator>
<numr-calculator autosave storage-key="calc-2"></numr-calculator>

// Custom WASM path (for CDN hosting)
<numr-calculator base-path="/assets/numr"></numr-calculator>

// Listen to events
<numr-calculator id="calc"></numr-calculator>
<script>
  document.getElementById('calc').addEventListener('numr-change', (e) => {
    console.log(e.detail.content);  // Current content
    console.log(e.detail.results);  // Evaluation results
    console.log(e.detail.totals);   // Running totals
  });
</script>

// Programmatic API
const calc = document.querySelector('numr-calculator');
calc.setContent('10 + 20');
calc.getContent();     // "10 + 20"
calc.clear();          // Clear document
calc.refreshRates();   // Fetch fresh exchange rates
calc.focus();          // Focus the editor
